--
-- --------------------------------------------------------
-- ENTIDADES INDEPENDIENTES
-- --------------------------------------------------------
--

CREATE TABLE alumno (
	dni VARCHAR(9) NOT NULL DEFAULT '0',
	nombre VARCHAR(20) NOT NULL DEFAULT '',
	apellidos VARCHAR(30) NOT NULL DEFAULT '',
	telefono VARCHAR(9) DEFAULT NULL,
 	poblacion VARCHAR(20) NOT NULL DEFAULT '',

	CONSTRAINT pk_alumno PRIMARY KEY(dni)
) ENGINE=InnoDB DEFAULT CHARSET=latin1;

-- --------------------------------------------------------

CREATE TABLE curso (
	codigo NUMERIC(11) NOT NULL DEFAULT '00000000000',
	nombre VARCHAR(50) NOT NULL DEFAULT '',

	CONSTRAINT pk_curso PRIMARY KEY(codigo),
	CONSTRAINT ck_codigo CHECK(codigo BETWEEN 00000000000 AND 99999999999),
	CONSTRAINT ck_nombre UNIQUE(nombre)
) ENGINE=InnoDB DEFAULT CHARSET=latin1;

-- --------------------------------------------------------

CREATE TABLE modulo (
	codigo NUMERIC(11) NOT NULL DEFAULT '00000000000',
	nombre VARCHAR(30) NOT NULL DEFAULT '',
	n_practicas NUMERIC(2) NOT NULL DEFAULT '00',
	n_examenes NUMERIC(2) NOT NULL DEFAULT '00',

	CONSTRAINT pk_modulo PRIMARY KEY(codigo),
	CONSTRAINT ck_n_practicas CHECK(n_practicas BETWEEN 00 AND 99),
	CONSTRAINT ck_n_examenes CHECK(n_examenes BETWEEN 00 AND 99)
) ENGINE=InnoDB DEFAULT CHARSET=latin1;

-- --------------------------------------------------------

CREATE TABLE trimestre (
	codigo NUMERIC(1) NOT NULL DEFAULT '0',

	CONSTRAINT pk_trimestre PRIMARY KEY(codigo),
	CONSTRAINT ck_codigo CHECK(codigo BETWEEN 0 AND 3)
) ENGINE=InnoDB DEFAULT CHARSET=latin1;


--
-- --------------------------------------------------------
-- TABLAS DE RELACIÓN
-- --------------------------------------------------------
--

CREATE TABLE cursa (
	id INT AUTO_INCREMENT,

	dni_alumno VARCHAR(9) NOT NULL DEFAULT '0',
	codigo_curso NUMERIC(11) NOT NULL DEFAULT '00000000000',
	anio VARCHAR(9) NOT NULL DEFAULT '0000-0000', -- Formato "2025-2026"
	nota DECIMAL(4, 2) NOT NULL DEFAULT 0.00,

	CONSTRAINT pk_cursa PRIMARY KEY(id),

	CONSTRAINT fk_cursa_alumno FOREIGN KEY (dni_alumno) REFERENCES alumno(dni),
	CONSTRAINT fk_cursa_curso FOREIGN KEY (codigo_curso) REFERENCES curso(codigo),

	CONSTRAINT ck_anio CHECK (anio LIKE '____-____'),
	CONSTRAINT ck_nota_cursa CHECK(nota BETWEEN 0.00 AND 10.00)
) ENGINE=InnoDB DEFAULT CHARSET=latin1;

-- --------------------------------------------------------

CREATE TABLE contiene (
	id INT AUTO_INCREMENT,
	
	codigo_curso NUMERIC(11) NOT NULL DEFAULT '00000000000',
	codigo_modulo NUMERIC(11) NOT NULL DEFAULT '00000000000',

	CONSTRAINT pk_contiene PRIMARY KEY(id),

	CONSTRAINT fk_cont_curso FOREIGN KEY (codigo_curso) REFERENCES curso(codigo),
	CONSTRAINT fk_cont_modulo FOREIGN KEY (codigo_modulo) REFERENCES modulo(codigo)
) ENGINE=InnoDB DEFAULT CHARSET=latin1;

-- --------------------------------------------------------

CREATE TABLE califica (
	id INT AUTO_INCREMENT,
	dni_alumno VARCHAR(9) NOT NULL DEFAULT '0',
	codigo_modulo NUMERIC(11) NOT NULL DEFAULT '00000000000',
	codigo_trimestre NUMERIC(1) NOT NULL DEFAULT '0',
	nota DECIMAL(4, 2) NOT NULL DEFAULT 0.00,
	tipo VARCHAR(20) NOT NULL DEFAULT 'Práctica', -- Solo 'Práctica' o 'Examen'
	
	CONSTRAINT pk_califica PRIMARY KEY(id),

	CONSTRAINT fk_cal_alumno FOREIGN KEY (dni_alumno) REFERENCES alumno(dni),
	CONSTRAINT fk_cal_modulo FOREIGN KEY (codigo_modulo) REFERENCES modulo(codigo),

	CONSTRAINT ck_trimestre CHECK(codigo_trimestre BETWEEN 0 AND 3),
	CONSTRAINT fk_cal_trimestre FOREIGN KEY (codigo_trimestre) REFERENCES trimestre(codigo),

	CONSTRAINT ck_nota_cal CHECK(nota BETWEEN 0.00 AND 10.00),	
	CONSTRAINT ck_tipo CHECK (tipo IN ('Práctica', 'Examen'))
) ENGINE=InnoDB DEFAULT CHARSET=latin1;
